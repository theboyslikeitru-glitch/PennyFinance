Firestore Security Rules для Penny\n===================================\n\nСкопируй этот текст в Firebase Console → Firestore Database → Rules, затем нажми Publish.\n\nrules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Каждый пользователь читает и меняет только собственные документы.
    match /users/{userId}/transactions/{transactionId} {
      allow read, delete: if request.auth != null
                          && request.auth.uid == userId;

      allow create: if request.auth != null
                    && request.auth.uid == userId
                    && validTransaction(request.resource.data);

      allow update: if request.auth != null
                    && request.auth.uid == userId
                    && validTransaction(request.resource.data);
    }

    // По умолчанию всё, что не описано выше, закрыто.
    match /{document=**} {
      allow read, write: if false;
    }

    function validTransaction(data) {
      return data.keys().hasAll([
               'type', 'amount', 'description', 'category',
               'date', 'account', 'note'
             ])
             && data.type in ['expense', 'income', 'transfer']
             && data.amount is number
             && data.amount > 0
             && data.amount <= 1000000000
             && data.description is string
             && data.description.size() > 0
             && data.description.size() <= 80
             && data.category is string
             && data.category.size() > 0
             && data.category.size() <= 50
             && data.date is string
             && data.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
             && data.account is string
             && data.account.size() <= 40
             && data.note is string
             && data.note.size() <= 180;
    }
  }
}
